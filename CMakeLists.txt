cmake_minimum_required(VERSION 3.22)
project(pycli VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)

configure_file(src/expr_spec.hpp.in generated/expr_spec.hpp @ONLY)
configure_file(src/worker.hpp.in generated/worker.hpp @ONLY)
add_executable(pycli src/main.cpp src/safetensors.cpp)

target_include_directories(pycli PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set_target_properties(pycli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
